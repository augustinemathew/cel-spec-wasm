# Host-only symbols exported by cel_runtime.wasm.
#
# These are the exports that are NOT codegen-imported helpers.  The
# codegen-helper export set is NO LONGER listed here — it is derived
# from the `// cel:codegen-export` markers in `runtime/cel_*.{h,c}` by
# `//bazel:gen_runtime_catalogue` (the C source is the single source of
# truth for the catalogue).
#
# Read at build time by `//runtime:wasm_export_args` (genrule below),
# which UNIONs the generated codegen-helper names with this host-only
# list and emits one `--export=<name>` line per symbol into a wasm-ld
# response file, consumed by `cel_runtime_wasm.bin` via `-Wl,@<rsp>`.
#
# Read at test time by `//abi:runtime_catalogue_consistency_test` — that
# test asserts the generated codegen-helper set and this host-only set
# are disjoint, and that the well-known system symbols below are present.
#
# Format:
#   - One symbol per line.  Whitespace-only and `#`-prefixed lines
#     ignored.
#   - A leading `[<name>]` section header is tolerated (the parser skips
#     it); only `[host-only]` exists now.
#
# Three sub-buckets:
#
#   - system / linker-internal: required for the wasi-sdk toolchain,
#     not codegen-referenced by name.
#
#   - public arena API for host reentry: the engine calls these
#     directly via the WasmtimeArenaAllocator and host-reentry path
#     (activation marshalling).  Distinct from the codegen-helper
#     `arena_alloc` / `arena_reset`, which codegen imports for the
#     comprehension / list-builder prologue.
#
#   - same-kind eq/ne fast paths + `_arena` tail-call targets:
#     internal to the wasm module's dispatcher pattern.  Codegen
#     calls the polymorphic dispatcher (`cel_equals_at_vv`,
#     `cel_list_size`, …); the dispatcher tail-calls into one of
#     these.  Exported so native-host tests (`//runtime:cel_*_test`)
#     link against them directly.

[host-only]

# Same-kind eq rungs of `equality_kernel` (cel_runtime.c) — exported
# so wat_runner fixtures and native tests can drive each rung directly
# without round-tripping through the polymorphic dispatcher.  (The
# historical per-kind int/uint/double eq+ne and bool_ne/numeric_ne
# arms were unreachable from compiled CEL — equality lowers only to
# the polymorphic `cel_equals_at_vv` / `cel_not_equals_at_vv` — and
# were deleted with their definitions.)
cel_bool_eq_at_vv

cel_bytes_eq_at_vv
